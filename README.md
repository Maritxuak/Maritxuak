<!--
═══════════════════════════════════════════════════════════════
  КАК ВКЛЮЧИТЬ SNAKE-АНИМАЦИЮ (один раз):

  1. В репо Maritxuak/Maritxuak создай файл:
     .github/workflows/snake.yml

  2. Вставь содержимое из блока ниже и сделай commit.

  3. Actions → Generate Snake → Run workflow
     (или дождись cron / push)

  4. Появится ветка `output` с SVG — змейка заработает.
═══════════════════════════════════════════════════════════════

# --- файл: .github/workflows/snake.yml ---
name: Generate Snake

on:
  schedule:
    - cron: "0 0 * * *"
  workflow_dispatch:
  push:
    branches:
      - main

jobs:
  generate:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    permissions:
      contents: write
    steps:
      - name: Generate snake
        uses: Platane/snk@v3
        with:
          github_user_name: Maritxuak
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark

      - name: Push to output branch
        uses: crazy-max/ghaction-github-pages@v3
        with:
          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
-->
